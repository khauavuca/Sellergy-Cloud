class Component extends DCLogic {
  constructor(props) { super(props); this.state = { modo: 'escolha', e: 1 }; }
  renderVals() {
    const s = this.state; const set = (o) => this.setState(o);
    const nomes = ['Facebook', 'Número', 'Código', 'Pronto'];
    return {
      escolha: s.modo === 'escolha', oficial: s.modo === 'oficial', qr: s.modo === 'qr',
      o1: s.e === 1, o2: s.e === 2, o3: s.e === 3, o4: s.e === 4, navOf: s.e === 2 || s.e === 3,
      passos: nomes.map((t, i) => ({ t, n: i + 1 < s.e ? '✓' : i + 1, cls: 'st' + (i + 1 === s.e ? ' on' : i + 1 < s.e ? ' done' : '') })),
      digitos: ['4', '8', '2', '0', '1', '7'],
      modelos: [
        { t: 'Lembrete 24 h com Confirmar e Remarcar', st: 'Aprovado', cls: 'pill ok' },
        { t: 'Lembrete 2 h', st: 'Aprovado', cls: 'pill ok' },
        { t: 'Cobrança de sinal por Pix', st: 'Em análise', cls: 'pill wait' },
        { t: 'Oferta de encaixe', st: 'Em análise', cls: 'pill wait' },
      ],
      irOficial: () => set({ modo: 'oficial', e: 1 }), irQr: () => set({ modo: 'qr' }), voltarEscolha: () => set({ modo: 'oficial', e: 1 }),
      prox: () => set({ e: Math.min(4, s.e + 1) }), voltar: () => set({ e: Math.max(1, s.e - 1) }),
    };
  }
}
