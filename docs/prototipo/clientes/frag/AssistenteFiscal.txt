class Component extends DCLogic {
  constructor(props) { super(props); this.state = { e: 1, nfse: true, nfce: true, cert: false, teste: 0 }; }
  renderVals() {
    const s = this.state; const set = (o) => this.setState(o);
    const nomes = ['Empresa', 'Certificado', 'Configuração', 'Teste'];
    const ok = s.teste === 2;
    const ck = (t, x) => ({ t, x: ok ? x : s.teste === 1 ? '...' : '', ic: ok ? '✓' : '•', cls: 'pill ' + (ok ? 'ok' : 'neu') });
    return {
      passos: nomes.map((t, i) => ({ t, n: i + 1 < s.e ? '✓' : i + 1, cls: 'st' + (i + 1 === s.e ? ' on' : i + 1 < s.e ? ' done' : '') })),
      e1: s.e === 1, e2: s.e === 2, e3: s.e === 3, e4: s.e === 4, navegando: s.e < 4,
      n1: 'chip' + (s.nfse ? ' on' : ''), n2: 'chip' + (s.nfce ? ' on' : ''),
      t1: () => set({ nfse: !s.nfse }), t2: () => set({ nfce: !s.nfce }),
      semCert: !s.cert, comCert: s.cert, enviarCert: () => set({ cert: true }),
      checks: [ck('Certificado aceito', '0,4 s'), ck('Prefeitura de São Paulo respondeu', '1,2 s'), ck('NFS-e de teste autorizada', 'nº 1'), ck('NFC-e de teste autorizada (SEFAZ-SP)', 'nº 1')],
      testando: !ok, ativo: ok,
      testar: () => set({ teste: 2 }),
      avancar: () => set({ e: s.e + 1, cert: s.e === 1 ? s.cert : true }), voltar: () => set({ e: Math.max(1, s.e - 1) }),
    };
  }
}
