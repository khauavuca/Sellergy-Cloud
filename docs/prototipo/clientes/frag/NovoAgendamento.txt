class Component extends DCLogic {
  constructor(props) { super(props); this.state = { e: 1, cli: 0, srv: 0, prof: 0, dia: 0, hora: 3, sinal: true }; }
  renderVals() {
    const s = this.state; const set = (o) => this.setState(o);
    const nomes = ['Cliente', 'Serviço', 'Horário', 'Confirmação'];
    const cl = [
      { nome: 'Paula Rocha', ini: 'PR', info: '(11) 99812-4410 · última visita 10/11 · pacote ativo' },
      { nome: 'Paulo Henrique Dias', ini: 'PH', info: '(11) 98120-3377 · última visita em agosto' },
      { nome: 'Ana Paula Moreira', ini: 'AP', info: '(11) 97455-0921 · cliente nova' },
    ];
    const sv = [
      { nome: 'Limpeza de pele', info: '1 h · Ana', preco: 180, p: [0] },
      { nome: 'Drenagem linfática', info: '50 min · Bruno', preco: 150, p: [1] },
      { nome: 'Peeling químico', info: '1 h · Carla', preco: 350, p: [2] },
      { nome: 'Massagem relaxante', info: '1 h · Bruno, Ana', preco: 160, p: [1, 0] },
    ];
    const pf = ['Ana Ribeiro', 'Bruno Teles', 'Carla Nunes'];
    const dias = ['Hoje, 24/11', 'Qua, 25/11', 'Qui, 26/11', 'Sex, 27/11', 'Sáb, 28/11'];
    const hs = ['09:00', '09:30', '10:00', '10:30', '11:00', '13:30', '14:00', '14:30', '15:00', '16:00', '16:30', '17:00'];
    const ocup = s.dia === 0 ? [0, 1, 2, 7] : [4, 9];
    const srv = sv[s.srv]; const profIdx = srv.p.includes(s.prof) ? s.prof : srv.p[0];
    const brl = (v) => 'R$ ' + v.toLocaleString('pt-BR', { minimumFractionDigits: 2 });
    const quando = dias[s.dia] + ' às ' + hs[s.hora];
    const pacote = s.srv === 1;
    const sinalV = pacote ? 0 : Math.round(srv.preco * 0.3);
    return {
      passos: nomes.map((t, i) => ({ n: i + 1 < s.e ? '✓' : i + 1, t, cls: 'st' + (i + 1 === s.e ? ' on' : i + 1 < s.e ? ' done' : '') })),
      e1: s.e === 1, e2: s.e === 2, e3: s.e === 3, e4: s.e === 4, e5: s.e === 5, navegando: s.e < 5,
      clientes: cl.map((c, i) => ({ ...c, cls: 'opt' + (s.cli === i ? ' on' : ''), pick: () => set({ cli: i }) })),
      servicos: sv.map((x, i) => ({ ...x, preco: brl(x.preco), cls: 'opt' + (s.srv === i ? ' on' : ''), pick: () => set({ srv: i, prof: x.p[0] }) })),
      profs: srv.p.map((i) => ({ nome: pf[i], cls: 'chip' + (profIdx === i ? ' on' : ''), pick: () => set({ prof: i }) })),
      dias: dias.map((t, i) => ({ t, cls: 'chip' + (s.dia === i ? ' on' : ''), pick: () => set({ dia: i, hora: s.dia === i ? s.hora : 5 }) })),
      slots: hs.map((t, i) => ({ t, cls: 'slot' + (ocup.includes(i) ? ' x' : s.hora === i ? ' on' : ''), pick: () => { if (!ocup.includes(i)) set({ hora: i }); } })),
      profNome: pf[profIdx].split(' ')[0],
      swSinal: 'sw' + (s.sinal && !pacote ? ' on' : ''),
      toggleSinal: () => set({ sinal: !s.sinal }),
      sinalValor: pacote ? 'não se aplica, usa o pacote' : brl(sinalV),
      rCliente: cl[s.cli].nome,
      rServico: s.e > 1 ? srv.nome : '—',
      rProf: s.e > 1 ? pf[profIdx] : '—',
      rQuando: s.e > 2 ? quando : '—',
      rValor: s.e > 1 ? (pacote ? 'Pacote (sessão 5 de 10)' : brl(srv.preco)) : '—',
      rSinal: s.e > 3 ? (s.sinal && !pacote ? brl(sinalV) + ' por Pix' : 'Sem sinal') : '—',
      fimTxt: cl[s.cli].nome + ' · ' + srv.nome + ' com ' + pf[profIdx] + ' · ' + quando + '. ' + (s.sinal && !pacote ? 'O Pix do sinal de ' + brl(sinalV) + ' foi enviado no WhatsApp. O horário fica amarelo na agenda até ela pagar.' : 'A confirmação já foi enviada no WhatsApp.'),
      avancarTxt: s.e === 4 ? 'Marcar horário' : 'Continuar',
      avancar: () => set({ e: s.e + 1 }),
      voltar: () => set({ e: Math.max(1, s.e - 1) }),
      reiniciar: () => set({ e: 1 }),
    };
  }
}
