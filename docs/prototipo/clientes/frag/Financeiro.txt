  constructor(props) { super(props); this.state = { per: 1, aba: 0, av: false, modal: false, luz: false, pagos: [] }; }
  renderVals() {
    const s = this.state; const set = (o) => this.setState(o);
    const brl = (v) => 'R$ ' + v.toLocaleString('pt-BR', { minimumFractionDigits: 2, maximumFractionDigits: 2 });
    const pagar = [

      { id: 5, t: 'Sellergy · plano Profissional', d: 'Vence 05/12 · fim do teste', v: 119 },
    ];
    if (s.luz) pagar.unshift({ id: 6, t: 'Conta de luz', d: 'Vence 30/11 · fixa · repete todo mês', v: 386.4 });
    const receber = [
      { t: 'Cartão de crédito a cair', d: '14 vendas · próximos 30 dias', v: 1962, st: 'A cair' },

    ];
    const abertas = pagar.filter((c) => !c.pago && !s.pagos.includes(c.id));
    const totPagar = abertas.reduce((a, c) => a + c.v, 0);
    const vend = [1240, 28940, 86320][s.per];
    const contas = s.aba === 0
      ? pagar.map((c) => { const pg = c.pago || s.pagos.includes(c.id); return { ic: '↑', icCls: 'av', t: c.t, d: c.d, v: brl(c.v), st: pg ? 'Pago' : 'A pagar', cls: 'pill ' + (pg ? 'ok' : c.d.includes('27/11') ? 'wait' : 'neu'), podePagar: !pg, pagar: () => set({ pagos: [...s.pagos, c.id] }) }; })
      : receber.map((c) => ({ ic: '↓', icCls: 'av', t: c.t, d: c.d, v: brl(c.v), st: c.st, cls: 'pill ' + (c.atr ? 'bad' : 'ok'), podePagar: false, pagar: () => {} }));
    return {
      periodos: ['Hoje', 'Novembro', 'Últimos 3 meses'].map((t, i) => ({ t, cls: 'chip' + (s.per === i ? ' on' : ''), pick: () => set({ per: i }) })),
