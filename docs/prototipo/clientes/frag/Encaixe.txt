class Component extends DCLogic {
  constructor(props) { super(props); this.state = { t: 'alerta', auto: false }; }
  renderVals() {
    const t = this.state.t;
    const serve = t === 'alerta' ? ['pill info', 'Serve pra 16:00'] : t === 'enviando' ? ['pill wait', 'Oferta enviada'] : null;
    const fila = [
      { ini: 'RC', nome: 'Renata Campos', quer: 'Peeling químico · Carla', quando: 'Hoje ou amanhã, à tarde', s: 1 },
      { ini: 'LM', nome: 'Larissa Mendes', quer: 'Peeling químico · qualquer profissional', quando: 'Qualquer dia', s: 1 },
      { ini: 'DS', nome: 'Daniela Souza', quer: 'Peeling químico · Carla', quando: 'Ter a qui, depois das 14h', s: 1 },
      { ini: 'TV', nome: 'Thiago Vieira', quer: 'Massagem relaxante · Bruno', quando: 'Sábado de manhã', s: 0 },
      { ini: 'GA', nome: 'Gabriela Alves', quer: 'Limpeza de pele · Ana', quando: 'Sexta, 27/11', s: 0 },
      { ini: 'MF', nome: 'Mônica Freitas', quer: 'Drenagem linfática · Bruno', quando: 'Semana que vem', s: 0 },
    ];
    const lista = fila.filter((f, i) => !(t === 'feito' && i === 0)).map((f, i) => {
      let cls = 'pill neu', st = 'Aguardando vaga';
      if (f.s && serve) { cls = serve[0]; st = serve[1]; }
      if (f.s && t === 'feito') { cls = 'pill neu'; st = 'Avisada: preenchido'; }
      return { ...f, cls, st };
    });
    return {
      alerta: t === 'alerta', enviando: t === 'enviando', preenchido: t === 'feito',
      fila: lista, total: lista.length,
      swAuto: 'sw' + (this.state.auto ? ' on' : ''),
      toggleAuto: () => this.setState({ auto: !this.state.auto }),
      oferecer: () => this.setState({ t: 'enviando' }),
      aceitar: () => this.setState({ t: 'feito' }),
      reiniciar: () => this.setState({ t: 'alerta' }),
    };
  }
}
