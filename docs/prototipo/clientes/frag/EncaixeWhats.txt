class Component extends DCLogic {
  constructor(props) { super(props); this.state = { r: '' }; }
  renderVals() {
    const r = this.state.r;
    return {
      pergunta: !r, aceitou: r === 'sim', recusou: r === 'nao', respondeu: !!r,
      quero: () => this.setState({ r: 'sim' }),
      nao: () => this.setState({ r: 'nao' }),
      reiniciar: () => this.setState({ r: '' }),
    };
  }
}
