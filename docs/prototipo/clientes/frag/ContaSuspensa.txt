class Component extends DCLogic {
  constructor(props) { super(props); this.state = { pago: false, modal: false }; }
  renderVals() {
    const s = this.state; const set = (o) => this.setState(o);
    return {
      bloq: !s.pago, liberado: s.pago, modal: s.modal,
      vendasSt: s.pago ? 'Liberadas' : 'Pausadas', vendasX: s.pago ? 'Pode vender normalmente' : 'Pague a fatura pra liberar',

      pdvCls: s.pago ? '' : 'apag',
      abrir: () => set({ modal: true }), fechar: () => set({ modal: false }),
      pagar: () => set({ modal: false, pago: true }), reiniciar: () => set({ pago: false }),
    };
  }
}
