class Component extends DCLogic {
  constructor(props) { super(props); this.state = { sel: 0, pagos: [] }; }
  renderVals() {
    const s = this.state; const set = (o) => this.setState(o);
    const brl = (v) => 'R$ ' + v.toLocaleString('pt-BR', { minimumFractionDigits: 2, maximumFractionDigits: 2 });
    const P = [
      { ini: 'CN', n: 'Carla Nunes', regra: '20% dos serviços · 5% dos produtos', vale: 370, linhas: [{ t: '35 peelings químicos', base: 'R$ 12.250,00', v: 2450 }, { t: 'Produtos vendidos', base: 'R$ 1.240,00', v: 62 }] },
      { ini: 'AR', n: 'Ana Ribeiro', regra: '15% dos serviços · 5% dos produtos', vale: 0, linhas: [{ t: '41 limpezas de pele', base: 'R$ 7.380,00', v: 1107 }, { t: '10 massagens', base: 'R$ 1.600,00', v: 240 }, { t: 'Produtos vendidos', base: 'R$ 220,00', v: 11 }] },
      { ini: 'BT', n: 'Bruno Teles', regra: 'R$ 35,00 por atendimento', vale: 0, linhas: [{ t: '22 drenagens', base: '', v: 770 }, { t: '10 massagens', base: '', v: 350 }] },
    ].map((p) => ({ ...p, tot: p.linhas.reduce((a, l) => a + l.v, 0) - p.vale }));
    const pago = (i) => s.pagos.includes(i);
    const pagoTot = P.filter((p, i) => pago(i)).reduce((a, p) => a + p.tot, 0);
    const sel = P[s.sel];
    return {
      profs: P.map((p, i) => ({ ...p, v: brl(p.tot), st: pago(i) ? 'Pago' : 'A pagar', stCls: 'pill ' + (pago(i) ? 'ok' : 'wait'), cls: 'opt' + (s.sel === i ? ' on' : ''), pick: () => set({ sel: i }) })),
      sel: { ...sel, v: brl(sel.tot), vale: sel.vale ? '− ' + brl(sel.vale) : 'R$ 0,00', linhas: sel.linhas.map((l) => ({ ...l, v: brl(l.v) })), aberto: !pago(s.sel), pago: pago(s.sel), primeiro: sel.n.split(' ')[0] },
      pagoTot: brl(pagoTot), pagoX: s.pagos.length + ' de 3 pessoas', faltaTot: brl(P.reduce((a, p) => a + p.tot, 0) - pagoTot),
      pagar: () => set({ pagos: [...s.pagos, s.sel] }),
    };
  }
}
