class Component extends DCLogic {
  constructor(props) { super(props); this.state = { e: 1, obj: 0, f: [true, true, false], modelo: 0, quando: 0 }; }
  renderVals() {
    const s = this.state; const set = (o) => this.setState(o);
    const nomes = ['Objetivo', 'Público', 'Mensagem', 'Envio'];
    const objs = [
      { t: 'Chamar de volta quem sumiu', d: 'Clientes sem visita há 60 dias ou mais' },
      { t: 'Parabéns com presente', d: 'Aniversariantes do mês, com cupom' },
      { t: 'Avisar de novidade', d: 'Serviço ou produto novo' },
      { t: 'Preencher horário vazio', d: 'Promoção pra dias e horários fracos' },
    ];
    const filtros = ['Sem visita há 60 dias ou mais', 'Gastou mais de R$ 100,00 na última visita', 'Só quem fez limpeza de pele'];
    let pub = 52; if (s.f[0]) pub = 44; if (s.f[1]) pub -= 6; if (s.f[2]) pub = Math.round(pub / 2);
    const podeAvancar = s.e !== 3 || s.modelo === 2;
    return {
      passos: nomes.map((t, i) => ({ t, n: i + 1 < s.e ? '✓' : i + 1, cls: 'st' + (i + 1 === s.e ? ' on' : i + 1 < s.e ? ' done' : '') })),
      e1: s.e === 1, e2: s.e === 2, e3: s.e === 3, e4: s.e === 4, e5: s.e === 5, navegando: s.e < 5,
      objetivos: objs.map((o, i) => ({ ...o, cls: 'opt' + (s.obj === i ? ' on' : ''), pick: () => set({ obj: i }) })),
      filtros: filtros.map((t, i) => ({ t, cls: 'sw' + (s.f[i] ? ' on' : ''), pick: () => { const a = [...s.f]; a[i] = !a[i]; set({ f: a }); } })),
      publico: pub, semOk: 4,
      modeloSt: ['Rascunho', 'Em análise', 'Aprovado pelo WhatsApp'][s.modelo], modeloCls: 'pill ' + ['neu', 'wait', 'ok'][s.modelo],
      analise: s.modelo === 1, precisaEnviar: s.modelo === 0,
      enviarAnalise: () => set({ modelo: 1 }), aprovar: () => set({ modelo: 2 }),
      quandos: [{ t: 'Hoje às 18:00', d: 'Horário com mais leitura pra clínicas' }, { t: 'Agora', d: 'Começa em 1 minuto' }].map((q, i) => ({ ...q, cls: 'opt' + (s.quando === i ? ' on' : ''), pick: () => set({ quando: i }) })),
      custo: 'R$ ' + (pub * 0.35).toFixed(2).replace('.', ','),
      avCls: podeAvancar ? 'bp' : 'bs',
      avancarTxt: s.e === 4 ? 'Agendar campanha' : s.e === 3 && !podeAvancar ? 'Espere a aprovação' : 'Continuar',
      avancar: () => { if (podeAvancar) set({ e: s.e + 1 }); },
      voltar: () => set({ e: Math.max(1, s.e - 1) }),
      reiniciar: () => set({ e: 1, modelo: 0 }),
    };
  }
}
